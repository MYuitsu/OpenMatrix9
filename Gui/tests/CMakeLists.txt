cmake_minimum_required(VERSION 3.24)
project(OpenMatrix9UiTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_executable(MatrixSidebarTest MatrixSidebarTest.cpp ../MatrixSidebar.cpp)
target_include_directories(MatrixSidebarTest PRIVATE ..)
target_compile_definitions(MatrixSidebarTest PRIVATE OM9_RESOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../../Resources")
target_link_libraries(MatrixSidebarTest PRIVATE Qt6::Widgets Qt6::Test "${CMAKE_CURRENT_SOURCE_DIR}/../../rust/target/release/openmatrix9_rust.lib" ws2_32 userenv ntdll bcrypt)
enable_testing()
add_test(NAME MatrixSidebarTest COMMAND MatrixSidebarTest)
set_tests_properties(MatrixSidebarTest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
