cmake_minimum_required(VERSION 3.25)
project(OM9StandaloneNativeProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(OM9_OPENNURBS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../build/standalone-deps/opennurbs" CACHE PATH "Pinned standalone openNURBS checkout")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${OM9_OPENNURBS_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE OM9_OPENNURBS_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT OM9_OPENNURBS_COMMIT STREQUAL "eb92af3ba1806b0a34a99aba0d3bda83e3d46083")
    message(FATAL_ERROR "Standalone openNURBS source pin mismatch")
endif()
# No legacy OM9ThreeDm, FreeCAD, Qt, OCCT, OCAF or Qym application target.
add_subdirectory("${OM9_OPENNURBS_SOURCE}" opennurbs EXCLUDE_FROM_ALL)
set_property(TARGET zlib PROPERTY INTERFACE_INCLUDE_DIRECTORIES "")
set(OM9_JSON_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../build/standalone-deps/nlohmann-json-3.11.3" CACHE PATH "Pinned JSON 3.11.3 single header and license")
file(SHA256 "${OM9_JSON_SOURCE}/json.hpp" OM9_JSON_HEADER_HASH)
file(SHA256 "${OM9_JSON_SOURCE}/LICENSE.MIT" OM9_JSON_LICENSE_HASH)
if(NOT OM9_JSON_HEADER_HASH STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6" OR
   NOT OM9_JSON_LICENSE_HASH STREQUAL "86b998c792894ccb911a1cb7994f7a9652894e7a094c0b5e45be2f553f45cf14")
    message(FATAL_ERROR "Standalone nlohmann/json 3.11.3 source/license pin mismatch")
endif()
add_executable(om9-exchange-native native_exchange.cpp)
target_include_directories(om9-exchange-native PRIVATE "${OM9_JSON_SOURCE}")
target_link_libraries(om9-exchange-native PRIVATE opennurbsStatic Bcrypt)
target_compile_definitions(om9-exchange-native PRIVATE NOMINMAX ON_CMAKE_BUILD)
add_executable(om9-opennurbs-probe opennurbs_probe.cpp)
target_link_libraries(om9-opennurbs-probe PRIVATE opennurbsStatic)
target_compile_definitions(om9-opennurbs-probe PRIVATE NOMINMAX ON_CMAKE_BUILD)
if(MSVC)
    include(ProcessorCount)
    ProcessorCount(OM9_BUILD_CPUS)
    math(EXPR OM9_BUILD_WORKERS "${OM9_BUILD_CPUS} * 60 / 100")
    if(OM9_BUILD_WORKERS LESS 1)
        set(OM9_BUILD_WORKERS 1)
    endif()
    message(STATUS "Standalone native compilation workers: ${OM9_BUILD_WORKERS} of ${OM9_BUILD_CPUS}")
    target_compile_options(opennurbsStatic PRIVATE "/MP${OM9_BUILD_WORKERS}")
    target_compile_options(om9-opennurbs-probe PRIVATE /W4 /WX /utf-8 /EHsc)
    target_link_options(om9-opennurbs-probe PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup)
    target_compile_options(om9-exchange-native PRIVATE /W4 /WX /utf-8 /EHsc)
    target_link_options(om9-exchange-native PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup)
endif()
