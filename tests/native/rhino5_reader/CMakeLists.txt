cmake_minimum_required(VERSION 3.24)
project(OM9Rhino5Reader LANGUAGES C CXX)
set(OM9_LEGACY_NURBS_SOURCE "" CACHE PATH "Verified unchanged McNeel SDK20130711 root")
if(NOT EXISTS "${OM9_LEGACY_NURBS_SOURCE}/opennurbs_pointcloud.cpp")
    message(FATAL_ERROR "Supply verified SDK20130711 source")
endif()
file(GLOB sources "${OM9_LEGACY_NURBS_SOURCE}/opennurbs_*.cpp" "${OM9_LEGACY_NURBS_SOURCE}/opennurbs_*.c")
file(GLOB zlib_sources "${OM9_LEGACY_NURBS_SOURCE}/zlib/*.c")
add_library(OM9LegacyNurbs STATIC ${sources} ${zlib_sources})
target_include_directories(OM9LegacyNurbs PUBLIC "${OM9_LEGACY_NURBS_SOURCE}")
target_compile_definitions(OM9LegacyNurbs PRIVATE ON_COMPILING_OPENNURBS _CRT_SECURE_NO_WARNINGS WIN64)
target_compile_definitions(OM9LegacyNurbs PUBLIC UNICODE _UNICODE)
if(MSVC)
    target_compile_options(OM9LegacyNurbs PUBLIC /UWIN32)
    target_compile_options(OM9LegacyNurbs PRIVATE /w)
    # VS2022 optimizer ICE in the unchanged2013 lookup table translation unit.
    set_source_files_properties("${OM9_LEGACY_NURBS_SOURCE}/opennurbs_lookup.cpp" PROPERTIES COMPILE_OPTIONS /Od)
endif()
add_executable(Rhino5PointCloudReader point_cloud_reader.cpp)
add_executable(Rhino5HatchReader hatch_reader.cpp)
target_link_libraries(Rhino5HatchReader PRIVATE OM9LegacyNurbs)
target_compile_features(Rhino5HatchReader PRIVATE cxx_std_17)
if(WIN32)
    target_link_libraries(Rhino5HatchReader PRIVATE rpcrt4)
endif()
target_link_libraries(Rhino5PointCloudReader PRIVATE OM9LegacyNurbs)
target_compile_features(Rhino5PointCloudReader PRIVATE cxx_std_17)
if(WIN32)
    target_link_libraries(Rhino5PointCloudReader PRIVATE rpcrt4)
endif()
