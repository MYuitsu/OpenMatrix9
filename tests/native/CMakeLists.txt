cmake_minimum_required(VERSION 3.24)
project(OM9ThreeDmTests LANGUAGES C CXX)
include(../../cmake/OpenNURBS.cmake)
enable_testing()
foreach(kind Geometry Preservation Inventory Archive Block)
    string(TOLOWER "${kind}" source_kind)
    set(test_kind "${kind}")
    if(kind STREQUAL "Block")
        set(source_kind blocks)
        set(test_kind Blocks)
    endif()
    add_executable(ThreeDm${kind}Tests three_dm_${source_kind}.cpp)
    target_link_libraries(ThreeDm${kind}Tests PRIVATE OM9ThreeDm)
    if(MSVC)
        target_compile_options(ThreeDm${kind}Tests PRIVATE /utf-8)
    endif()
    add_test(NAME OM9-FILE-012.${test_kind} COMMAND ThreeDm${kind}Tests)
endforeach()

set(OM9_RING_FIXTURE "" CACHE FILEPATH "Optional user-supplied ring model; not distributed")
if(EXISTS "${OM9_RING_FIXTURE}")
    add_executable(ThreeDmRingTests three_dm_ring.cpp)
    target_link_libraries(ThreeDmRingTests PRIVATE OM9ThreeDm)
    target_compile_definitions(ThreeDmRingTests PRIVATE OM9_RING_FIXTURE=\"${OM9_RING_FIXTURE}\")
    add_test(NAME OM9-FILE-012.UserRing COMMAND ThreeDmRingTests)
endif()
set(OM9_UNMESHED_FIXTURE "" CACHE FILEPATH "Optional user-supplied unmeshed model; not distributed")
if(EXISTS "${OM9_UNMESHED_FIXTURE}")
    add_executable(ThreeDmUnmeshedTests three_dm_unmeshed.cpp)
    target_link_libraries(ThreeDmUnmeshedTests PRIVATE OM9ThreeDm)
    target_compile_definitions(ThreeDmUnmeshedTests PRIVATE OM9_UNMESHED_FIXTURE=\"${OM9_UNMESHED_FIXTURE}\")
    add_test(NAME OM9-FILE-012.Unmeshed COMMAND ThreeDmUnmeshedTests)
endif()
