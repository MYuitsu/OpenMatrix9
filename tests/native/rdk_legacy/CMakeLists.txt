cmake_minimum_required(VERSION 3.24)
project(OM9RdkLegacyReader LANGUAGES CXX)
set(OM9_LEGACY_NURBS_SOURCE "" CACHE PATH "Unchanged SDK2013 headers")
set(OM9_LEGACY_NURBS_LIBRARY "" CACHE FILEPATH "Independent unchanged SDK2013 library")
if(NOT EXISTS "${OM9_LEGACY_NURBS_SOURCE}/opennurbs.h" OR NOT EXISTS "${OM9_LEGACY_NURBS_LIBRARY}")
    message(FATAL_ERROR "Supply the independently verified SDK2013 headers and library")
endif()
add_library(legacy STATIC IMPORTED)
set_target_properties(legacy PROPERTIES IMPORTED_LOCATION "${OM9_LEGACY_NURBS_LIBRARY}")
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(RdkLegacyReader rdk_legacy_reader.cpp)
target_include_directories(RdkLegacyReader PRIVATE "${OM9_LEGACY_NURBS_SOURCE}")
target_compile_definitions(RdkLegacyReader PRIVATE UNICODE _UNICODE)
target_compile_options(RdkLegacyReader PRIVATE /UWIN32 /utf-8)
target_compile_features(RdkLegacyReader PRIVATE cxx_std_17)
target_link_libraries(RdkLegacyReader PRIVATE legacy Qt6::Core rpcrt4)
