cmake_minimum_required(VERSION 3.24)
project(OM9ModelingLegacy LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
set(OM9_LEGACY_INCLUDE "" CACHE PATH "Verified unchanged SDK20130711 source")
set(OM9_LEGACY_LIBRARY "" CACHE FILEPATH "Existing unchanged SDK20130711 static library")
add_library(ModelingLegacyNurbs STATIC IMPORTED)
set_target_properties(ModelingLegacyNurbs PROPERTIES IMPORTED_LOCATION "${OM9_LEGACY_LIBRARY}")
add_executable(ModelingLegacyReader ../modeling_legacy_reader.cpp)
target_include_directories(ModelingLegacyReader PRIVATE "${OM9_LEGACY_INCLUDE}")
target_compile_features(ModelingLegacyReader PRIVATE cxx_std_17)
target_compile_definitions(ModelingLegacyReader PRIVATE UNICODE _UNICODE)
target_compile_options(ModelingLegacyReader PRIVATE /UWIN32 /utf-8)
target_link_libraries(ModelingLegacyReader PRIVATE ModelingLegacyNurbs Qt6::Core rpcrt4)
