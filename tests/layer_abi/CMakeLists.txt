cmake_minimum_required(VERSION 3.24)
project(OM9LayerAbi LANGUAGES CXX)
enable_testing()
get_filename_component(OM9_MODULE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
set(RUST_TARGET "${CMAKE_CURRENT_BINARY_DIR}/rust-target")
set(RUST_LIBRARY "${RUST_TARGET}/release/openmatrix9_rust.lib")
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "${OM9_MODULE}/rust/src/*.rs")
add_custom_command(OUTPUT "${RUST_LIBRARY}"
    COMMAND "${CARGO_EXECUTABLE}" build --release --locked
        --manifest-path "${OM9_MODULE}/rust/Cargo.toml" --target-dir "${RUST_TARGET}"
    DEPENDS ${RUST_SOURCES} "${OM9_MODULE}/rust/Cargo.toml" VERBATIM)
add_custom_target(OM9LayerRust DEPENDS "${RUST_LIBRARY}")
add_library(OM9LayerRustLib STATIC IMPORTED GLOBAL)
set_target_properties(OM9LayerRustLib PROPERTIES IMPORTED_LOCATION "${RUST_LIBRARY}")
add_executable(LayerRustAbiTests "${OM9_MODULE}/tests/native/layer_rust_abi.cpp")
add_dependencies(LayerRustAbiTests OM9LayerRust)
target_compile_features(LayerRustAbiTests PRIVATE cxx_std_23)
target_include_directories(LayerRustAbiTests PRIVATE "${OM9_MODULE}/Gui")
target_link_libraries(LayerRustAbiTests PRIVATE OM9LayerRustLib ws2_32 userenv ntdll bcrypt)
target_compile_options(LayerRustAbiTests PRIVATE /utf-8)
add_test(NAME OM9-LAYER-001.RustAbi COMMAND LayerRustAbiTests)

# Reuse the accepted pinned native SDK library read-only for metadata fixtures.
# Full module/native regressions are a separate gate, not replaced by this target.
option(OM9_LAYER_NATIVE_TESTS "Build native 3DM layer metadata tests" OFF)
set(OM9_OPENNURBS_SOURCE "" CACHE PATH "Accepted pinned openNURBS headers")
set(OM9_OPENNURBS_LIBRARY "" CACHE FILEPATH "Accepted pinned static library")
set(OM9_OPENNURBS_ZLIB "" CACHE FILEPATH "Matching SDK zlib static library")
if(OM9_LAYER_NATIVE_TESTS)
    foreach(required OM9_OPENNURBS_SOURCE OM9_OPENNURBS_LIBRARY OM9_OPENNURBS_ZLIB)
        if(NOT EXISTS "${${required}}")
            message(FATAL_ERROR "Missing read-only SDK ${required}")
        endif()
    endforeach()
    add_executable(LayerExchangeNativeTests "${OM9_MODULE}/tests/native/layer_exchange_native.cpp" "${OM9_MODULE}/Gui/LayerExchangeAdapter.cpp")
    add_dependencies(LayerExchangeNativeTests OM9LayerRust)
    target_compile_features(LayerExchangeNativeTests PRIVATE cxx_std_23)
    target_compile_definitions(LayerExchangeNativeTests PRIVATE NOMINMAX OM9_LAYER_FIXTURES="${CMAKE_CURRENT_BINARY_DIR}/layer-fixtures")
    target_compile_options(LayerExchangeNativeTests PRIVATE /utf-8)
    target_include_directories(LayerExchangeNativeTests PRIVATE "${OM9_MODULE}/Gui" "${OM9_OPENNURBS_SOURCE}")
    target_link_libraries(LayerExchangeNativeTests PRIVATE OM9LayerRustLib "${OM9_OPENNURBS_LIBRARY}" "${OM9_OPENNURBS_ZLIB}" Shlwapi Usp10 ws2_32 userenv ntdll bcrypt)
    add_test(NAME OM9-LAYER-001.NativePalette COMMAND LayerExchangeNativeTests)
    add_executable(LayerRetainedNativeTests "${OM9_MODULE}/tests/native/layer_retained_native.cpp" "${OM9_MODULE}/Gui/LayerExchangeAdapter.cpp")
    add_dependencies(LayerRetainedNativeTests OM9LayerRust)
    target_compile_features(LayerRetainedNativeTests PRIVATE cxx_std_23)
    target_compile_definitions(LayerRetainedNativeTests PRIVATE NOMINMAX OM9_LAYER_FIXTURES="${CMAKE_CURRENT_BINARY_DIR}/retained-layer-fixtures")
    target_compile_options(LayerRetainedNativeTests PRIVATE /utf-8)
    target_include_directories(LayerRetainedNativeTests PRIVATE "${OM9_MODULE}/Gui" "${OM9_OPENNURBS_SOURCE}")
    target_link_libraries(LayerRetainedNativeTests PRIVATE OM9LayerRustLib "${OM9_OPENNURBS_LIBRARY}" "${OM9_OPENNURBS_ZLIB}" Shlwapi Usp10 ws2_32 userenv ntdll bcrypt)
    add_test(NAME OM9-LAYER-001.NativeRetainedPalette COMMAND LayerRetainedNativeTests)
endif()

option(OM9_LAYER_ARCHIVE_TESTS "Build real archive integration tests using the accepted converter library" OFF)
set(OM9_NATIVE_ARCHIVE_LIBRARY "" CACHE FILEPATH "Accepted geometry conversion library")
if(OM9_LAYER_ARCHIVE_TESTS)
    if(NOT OM9_LAYER_NATIVE_TESTS OR NOT EXISTS "${OM9_NATIVE_ARCHIVE_LIBRARY}")
        message(FATAL_ERROR "Archive tests need the pinned native libraries")
    endif()
    find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)
    find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
    add_library(OM9LayerArchiveCandidate STATIC "${OM9_MODULE}/Gui/ThreeDmArchive.cpp"
        "${OM9_MODULE}/Gui/LayerExchangeAdapter.cpp" "${OM9_MODULE}/Gui/ThreeDmStaging.cpp"
        "${OM9_MODULE}/Gui/ThreeDmModeling.cpp")
    add_dependencies(OM9LayerArchiveCandidate OM9LayerRust)
    target_compile_features(OM9LayerArchiveCandidate PUBLIC cxx_std_23)
    target_compile_definitions(OM9LayerArchiveCandidate PUBLIC NOMINMAX)
    target_compile_options(OM9LayerArchiveCandidate PRIVATE /utf-8 /bigobj)
    target_include_directories(OM9LayerArchiveCandidate PUBLIC "${OM9_MODULE}/Gui" "${OM9_OPENNURBS_SOURCE}" ${OpenCASCADE_INCLUDE_DIR})
    target_link_libraries(OM9LayerArchiveCandidate PUBLIC OM9LayerRustLib "${OM9_NATIVE_ARCHIVE_LIBRARY}"
        "${OM9_OPENNURBS_LIBRARY}" "${OM9_OPENNURBS_ZLIB}" Qt6::Core
        TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKShHealing
        Shlwapi Usp10 ws2_32 userenv ntdll bcrypt)
    add_executable(LayerArchiveTests "${OM9_MODULE}/tests/native/layer_archive.cpp")
    target_compile_definitions(LayerArchiveTests PRIVATE OM9_LAYER_FIXTURES="${CMAKE_CURRENT_BINARY_DIR}/archive-fixtures")
    target_link_libraries(LayerArchiveTests PRIVATE OM9LayerArchiveCandidate)
    add_test(NAME OM9-LAYER-001.NativeArchive COMMAND LayerArchiveTests)
    add_executable(LayerStagingTests "${OM9_MODULE}/tests/native/layer_staging.cpp")
    target_compile_definitions(LayerStagingTests PRIVATE OM9_LAYER_FIXTURES="${CMAKE_CURRENT_BINARY_DIR}/archive-fixtures")
    target_link_libraries(LayerStagingTests PRIVATE OM9LayerArchiveCandidate)
    add_test(NAME OM9-LAYER-001.NativeStaging COMMAND LayerStagingTests)
    add_executable(LayerClipboardTests "${OM9_MODULE}/tests/native/layer_clipboard.cpp" "${OM9_MODULE}/Gui/LayerClipboard.cpp")
    target_compile_definitions(LayerClipboardTests PRIVATE OM9_LAYER_FIXTURES="${CMAKE_CURRENT_BINARY_DIR}/archive-fixtures")
    target_link_libraries(LayerClipboardTests PRIVATE OM9LayerArchiveCandidate Qt6::Widgets ole32 user32)
    add_test(NAME OM9-LAYER-001.NativeClipboard COMMAND LayerClipboardTests)
    set_tests_properties(OM9-LAYER-001.NativeClipboard PROPERTIES DEPENDS OM9-LAYER-001.NativeArchive RUN_SERIAL TRUE)
    set_tests_properties(OM9-LAYER-001.NativeStaging PROPERTIES DEPENDS OM9-LAYER-001.NativeArchive)
    foreach(pair "LayerLegacyArchiveTests;three_dm_archive.cpp;OM9-FILE-012.LayerLegacyArchive" "LayerLegacyBlockTests;three_dm_blocks.cpp;OM9-FILE-012.LayerLegacyBlocks")
        list(GET pair 0 target)
        list(GET pair 1 filename)
        list(GET pair 2 test)
        add_executable(${target} "${OM9_MODULE}/tests/native/${filename}")
        target_compile_options(${target} PRIVATE /utf-8)
        target_link_libraries(${target} PRIVATE OM9LayerArchiveCandidate)
        add_test(NAME ${test} COMMAND ${target})
    endforeach()
    if(WIN32)
        get_filename_component(OM9_DEPENDENCY_BIN "${OpenCASCADE_DIR}/../../../bin" ABSOLUTE)
        set_tests_properties(OM9-LAYER-001.NativeArchive OM9-LAYER-001.NativeStaging OM9-LAYER-001.NativeClipboard
            OM9-FILE-012.LayerLegacyArchive OM9-FILE-012.LayerLegacyBlocks PROPERTIES
            ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${OM9_DEPENDENCY_BIN}")
        set_tests_properties(OM9-LAYER-001.NativeClipboard PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM_PLUGIN_PATH=${OM9_DEPENDENCY_BIN}/../lib/qt6/plugins/platforms")
    endif()
endif()
